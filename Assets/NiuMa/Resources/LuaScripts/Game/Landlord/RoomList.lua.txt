-- 描述：斗地主公开房列表（走 HTTP /player/game/landlord/public）
local base = require 'BehaviourBase';
local RoomItem = require "Game.Landlord.RoomItem";
local CommonUtils = require 'Common.CommonUtils';
local rapidjson = require('rapidjson');

local _module = setmetatable({}, { __index = base });

function _module:Awake()
    local obj = self.gameObject;
    self._itemList = {};
    local child = obj.transform:Find("Viewport/Content");
    if (child ~= nil) then
        self._itemParent = child;
    end
    self._requesting = false;
    self._refreshElapsed = 0.0;
end

function _module:Start()
    self:RefreshList();
end

function _module:Update()
    self._refreshElapsed = self._refreshElapsed + CS.UnityEngine.Time.unscaledDeltaTime;
    if (self._refreshElapsed > 10.0) then
        self._refreshElapsed = 0.0;
        if (CommonUtils.IsStringNotEmpty(CS.NiuMa.GameManager.Instance.VenueId)) then
            return;
        end
        self:RefreshList();
    end
end

function _module:SetEmptyTip(obj)
    self._emptyTip = obj;
end

function _module:SetEnterCallback(callback)
    self._enterCallback = callback;
end

function _module:RefreshList()
    if (self._requesting) then
        return;
    end
    self._requesting = true;
    CS.NiuMa.GameManager.Instance:AuthGet("/player/game/landlord/public", self.OnGetRoomsResponse);
end

function _module.OnGetRoomsResponse(code, text)
    local self = _module;
    self._requesting = false;
    if (code ~= 200) then
        print("[Landlord] 房间列表请求失败 code=" .. tostring(code));
        return;
    end
    local ok, t = pcall(rapidjson.decode, text or "{}");
    if (not ok or t == nil) then
        return;
    end
    local items = t.items or {};
    local prefab = CS.NiuMa.ResourceManager.Instance:LoadResource(
        "Prefabs/Game/Landlord/RoomItem", "prefabs/landlord.ab", "Assets/NiuMa/Resources/", ".prefab");
    if (prefab == nil) then
        print("[Landlord] RoomItem 预制体加载失败");
        return;
    end
    if (self._emptyTip ~= nil) then
        self._emptyTip:SetActive(#items < 1);
    end
    local GameObject = CS.UnityEngine.GameObject;
    local tmpList = {};
    for i = 1, #items do
        local item = self._itemList[i];
        if (item == nil) then
            local obj = GameObject.Instantiate(prefab, self._itemParent);
            item = RoomItem.NewInstance();
            item.gameObject = obj;
            item:Awake();
            item:SetEnterCallback(self._enterCallback);
        end
        item:SetRoomItem(items[i]);
        table.insert(tmpList, item);
    end
    -- 删除多余项
    for i = #items + 1, #self._itemList do
        local extra = self._itemList[i];
        if (extra ~= nil and extra.gameObject ~= nil) then
            CS.UnityEngine.Object.Destroy(extra.gameObject);
        end
    end
    self._itemList = tmpList;
end

return _module;
